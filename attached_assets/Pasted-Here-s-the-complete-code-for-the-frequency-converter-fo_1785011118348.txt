// active-app.js

const express = require('express');
const multer = require('multer');
const admZip = require('adm-zip');
const plist = require('plist');
const fs = require('fs');
const path = require('path');

const app = express();
const upload = multer({ dest: '/tmp/' });

app.use(express.json());
app.use(express.urlencoded({ extended: true }));

// Upload endpoint for IPA files
app.post('/upload', upload.single('ipa'), (req, res) => {
    const ipaPath = req.file.path;

    // Extract IPA file
    const zip = new admZip(ipaPath);
    const extractedPath = path.join('/tmp/', req.file.originalname.replace('.ipa', '/'));
    zip.extractAllTo(extractedPath, true);

    // Navigating to Payload and reading Info.plist
    const appBundlePath = path.join(extractedPath, 'Payload', fs.readdirSync(path.join(extractedPath, 'Payload'))[0]); // get the first *.app directory
    const plistPath = path.join(appBundlePath, 'Info.plist');

    if (!fs.existsSync(plistPath)) {
        return res.status(400).json({ error: 'Info.plist not found in the IPA.' });
    }

    // Parse Info.plist
    const plistData = plist.parse(fs.readFileSync(plistPath, 'utf8'));
    
    // Report checks
    const checks = {
        CFBundleDisplayName: {
            result: Boolean(plistData.CFBundleDisplayName && plistData.CFBundleDisplayName.length <= 30),
            explanation: "App name must be present and under 30 characters long."
        },
        CFBundleIdentifier: {
            result: Boolean(plistData.CFBundleIdentifier),
            explanation: "Bundle ID must be present."
        },
        CFBundleShortVersionString: {
            result: Boolean(plistData.CFBundleShortVersionString),
            explanation: "Short version string must be present."
        },
        CFBundleVersion: {
            result: Boolean(plistData.CFBundleVersion),
            explanation: "Build version number must be present."
        },
        MinimumOSVersion: {
            result: Boolean(plistData.MinimumOSVersion),
            explanation: "Minimum iOS version must be declared."
        },
        ITSAppUsesNonExemptEncryption: {
            result: Boolean(plistData.ITSAppUsesNonExemptEncryption),
            explanation: "Must be present for export compliance."
        },
        NSCameraUsageDescription: {
            result: Boolean(plistData.NSCameraUsageDescription),
            explanation: "Required if the app uses the camera."
        },
        NSMicrophoneUsageDescription: {
            result: Boolean(plistData.NSMicrophoneUsageDescription),
            explanation: "Required if the app uses the microphone."
        },
        NSPhotoLibraryUsageDescription: {
            result: Boolean(plistData.NSPhotoLibraryUsageDescription),
            explanation: "Required if the app accesses the photo library."
        },
        NSLocationWhenInUseUsageDescription: {
            result: Boolean(plistData.NSLocationWhenInUseUsageDescription),
            explanation: "Required if the app accesses location."
        },
        NSUserTrackingUsageDescription: {
            result: Boolean(plistData.NSUserTrackingUsageDescription),
            explanation: "Required for App Tracking Transparency."
        },
        AppTransportSecurity: {
            result: !plistData.NSAllowsArbitraryLoads,
            explanation: "Flag if NSAllowsArbitraryLoads is true."
        },
        IconFiles: {
            result: Boolean(plistData.CFBundleIcons),
            explanation: "App icons must be declared in the bundle."
        },
        UIBackgroundModes: {
            result: Boolean(plistData.UIBackgroundModes),
            explanation: "Any background modes must be declared."
        }
    };

    // Create the response
    let response = {
        success: true,
        checks: {}
    };

    for (const [key, check] of Object.entries(checks)) {
        response.checks[key] = {
            passed: check.result,
            explanation: check.explanation
        };
        if (!check.result) {
            response.success = false;
        }
    }

    res.json(response);
});

// Start server
const PORT = process.env.PORT || 3000;
app.listen(PORT, () => {
    console.log(`Server is running on port ${PORT}`);
}); 
Explanation of the Code:
Multer is used for handling file uploads.
adm-zip is used for extracting the IPA file.
plist is used to read and parse the Info.plist file within the IPA.
The app is set to check various properties from the Info.plist file and return a report on their status.
